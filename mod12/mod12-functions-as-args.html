Functions as Arguments

Function pointers, functors, lambdas, and std::function as algorithm callbacks.

The Big Picture

Many algorithms in the C++ Standard Library take functions or callable objects as parameters. These callables are often used as predicates or comparison functions to customize algorithm behavior.

There are several ways to send functions themselves as arguments to other functions, including function pointers, functors, lambda functions, and any callable target wrapped in a std::function object.

Function Pointer

#include <iostream>
#include <vector>
#include <algorithm>

// Function to check if a number is greater than 3
bool is_greater_than_three(int x)
{
    return x > 3;
}

int main()
{
    std::vector<int> data = {1, 2, 3, 4, 5};

    // Use std::find_if with a function pointer
    auto it = std::find_if(data.begin(), data.end(), is_greater_than_three);

    if (it != data.end())
    {
        std::cout << "Found a number greater than 3 using function pointer: " << *it << std::endl;
    }
    else
    {
        std::cout << "No number greater than 3 found using function pointer" << std::endl;
    }

    return 0;
}

Functor

#include <iostream>
#include <vector>
#include <algorithm>

struct GT_3
{
    bool operator()(int x) const
    {
        return x > 3;
    }
};

int main()
{
    std::vector<int> data = {1, 2, 3, 4, 5};

    // Use std::find_if with a functor
    auto it = std::find_if(data.begin(), data.end(), GT_3{});

    if (it != data.end())
    {
        std::cout << "Found a number greater than 3 using functor: " << *it << std::endl;
    }
    else
    {
        std::cout << "No number greater than 3 found using functor" << std::endl;
    }

    return 0;
}

Lambda Function

#include <iostream>
#include <vector>
#include <algorithm>

int main()
{
    std::vector<int> data = {1, 2, 3, 4, 5};

    // Use std::find_if with a lambda function
    auto it = std::find_if(data.begin(), data.end(), [](int x)
    {
        return x > 3;
    });

    if (it != data.end())
    {
        std::cout << "Found a number greater than 3 using lambda function: " << *it << std::endl;
    }
    else
    {
        std::cout << "No number greater than 3 found using lambda function" << std::endl;
    }

    return 0;
}

std::function

#include <iostream>
#include <vector>
#include <algorithm>
#include <functional>

int main()
{
    std::vector<int> data = {1, 2, 3, 4, 5};

    // Use std::find_if with std::function
    std::function<bool(int)> func = [](int x)
    {
        return x > 3;
    };

    auto it = std::find_if(data.begin(), data.end(), func);

    if (it != data.end())
    {
        std::cout << "Found a number greater than 3 using std::function: " << *it << std::endl;
    }
    else
    {
        std::cout << "No number greater than 3 found using std::function" << std::endl;
    }

    return 0;
}
